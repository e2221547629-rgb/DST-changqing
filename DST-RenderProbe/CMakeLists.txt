cmake_minimum_required(VERSION 3.21)
project(DSTRenderProbe LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "This prototype is Windows only.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "DSTRenderProbe requires Windows x64. Configure with -A x64.")
endif()
if(NOT DEFINED RESHADE_INCLUDE_DIR OR NOT EXISTS "${RESHADE_INCLUDE_DIR}/reshade.hpp")
    message(FATAL_ERROR "Set -DRESHADE_INCLUDE_DIR=C:/path/to/reshade/include (must contain reshade.hpp)")
endif()

add_library(DSTRenderProbe SHARED src/dst_render_probe.cpp)
target_compile_features(DSTRenderProbe PRIVATE cxx_std_17)
target_compile_definitions(DSTRenderProbe PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(DSTRenderProbe PRIVATE "${RESHADE_INCLUDE_DIR}")
target_link_libraries(DSTRenderProbe PRIVATE user32)
set_target_properties(DSTRenderProbe PROPERTIES PREFIX "" SUFFIX ".addon64")
if(MSVC)
    set_property(TARGET DSTRenderProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(DSTRenderProbe PRIVATE /W4 /EHsc /utf-8)
else()
    target_compile_options(DSTRenderProbe PRIVATE -Wall -Wextra)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(addon_smoke tests/addon_smoke.cpp)
    target_compile_features(addon_smoke PRIVATE cxx_std_17)
    target_compile_definitions(addon_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(addon_smoke PRIVATE "${RESHADE_INCLUDE_DIR}")
    if(MSVC)
        set_property(TARGET addon_smoke PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(addon_smoke PRIVATE /W4 /EHsc /utf-8)
    endif()
    add_test(NAME addon_smoke COMMAND addon_smoke "$<TARGET_FILE:DSTRenderProbe>")
    add_executable(probe_state_tests tests/probe_state_tests.cpp)
    target_compile_features(probe_state_tests PRIVATE cxx_std_17)
    if(MSVC)
        set_property(TARGET probe_state_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(probe_state_tests PRIVATE /W4 /EHsc /utf-8)
    endif()
    add_test(NAME probe_state_tests COMMAND probe_state_tests)
    add_executable(frame_capture_tests tests/frame_capture_tests.cpp)
    add_executable(d3d9_capture_smoke tests/d3d9_capture_smoke.cpp)
    target_link_libraries(d3d9_capture_smoke PRIVATE d3d9 user32)
    foreach(test_target frame_capture_tests d3d9_capture_smoke)
        target_compile_features(${test_target} PRIVATE cxx_std_17)
        target_compile_definitions(${test_target} PRIVATE NOMINMAX)
        if(MSVC)
            set_property(TARGET ${test_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            target_compile_options(${test_target} PRIVATE /W4 /EHsc /utf-8)
        endif()
        add_test(NAME ${test_target} COMMAND ${test_target})
    endforeach()
    set_tests_properties(d3d9_capture_smoke PROPERTIES SKIP_RETURN_CODE 77)
endif()
